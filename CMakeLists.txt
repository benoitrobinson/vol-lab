cmake_minimum_required(VERSION 3.26)
project(vollab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python 3.14 REQUIRED COMPONENTS Interpreter Development.Module)
find_package(nanobind CONFIG REQUIRED)
nanobind_add_module(_probe cpp/src/probe.cpp)
target_include_directories(_probe PRIVATE cpp/include)
install(TARGETS _probe LIBRARY DESTINATION vollab)
